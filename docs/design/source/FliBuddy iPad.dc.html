Landscape on a bench stand, usually with a keyboard. NavigationSplitView: the sidebar carries both connection states, Settings and Clear; the detail is the feed with every row on one line. Portrait collapses to the iPhone layout.

NavigationSplitView {
  List(selection: $pane) {
    Section("Connections") {
      FlipperStatusRow()
      BambuddyStatusRow() }
    NavigationLink(value: .feed) {
      Label("Feed", systemImage:
        "list.bullet") }.badge(feed.count)
    NavigationLink(value: .settings) {
      Label("Settings", systemImage:
        "gearshape") }
  }
  .safeAreaInset(edge: .bottom) {
    Button("Clear feed",
      systemImage: "trash",
      role: .destructive)
      .keyboardShortcut(.delete) }
} detail: {
  FeedTable()   // List, one-line rows
}

// same TabView as iPhone
