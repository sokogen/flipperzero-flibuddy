Apple Watch

Hands stay free: the spool in one, the Flipper in the other, the wrist does the rest. The Digital Crown sets the weight, one tap saves, a haptic confirms. Settings are read-only and arrive from the iPhone. Shown at 46 mm, enlarged 1.6×; watchOS is dark only.

NavigationStack {
  List(feed) { WatchFeedRow($0) }
    .navigationTitle("FliBuddy")
    .toolbar { ToolbarItem(
      placement: .topBarLeading) {
        ConnectionGlyph() } }
    .navigationDestination(for:) {
      RowDetail($0) }
}
// RowDetail
Text(grams, format: .number)
  .font(.system(size: 50,
    weight: .semibold,
    design: .rounded))
  .monospacedDigit()
  .focusable()
  .digitalCrownRotation($grams,
    from: 0, through: 5000, by: 1,
    sensitivity: .low,
    isHapticFeedbackEnabled: true)
// 5 g per detent above ~6 detents/s
