iPhone

TabView with Feed, Spools and Settings. Bluetooth to the Flipper. One-handed at the bench: the number pad rises on its own with the suggestion selected, Save stays above it, and the result is readable from a metre away.

TabView {
  Tab("Feed", systemImage: "list.bullet") {
    NavigationStack { FeedList()
      // List { ForEach(feed) {
      //   Section { FeedRow($0) } } }
      .navigationTitle("Feed")
      .toolbar { Button("Clear",
          systemImage: "trash") }
      .safeAreaInset(edge: .top) {
          ConnectionBanner() }
      .overlay { ContentUnavailableView }
      .sheet(item: $newRow) {
          NewSpoolForm() }
      .sheet(item: $linkRow) {
          LinkPicker().searchable(…) }
      .confirmationDialog(…) }
  }
  Tab("Spools", systemImage: "shippingbox") {
    NavigationStack { SpoolList()       // read-only
      .navigationTitle("Spools")
      .searchable(text: $query, prompt:
          "Material, colour, brand or note")
      .toolbar { Menu { sort; material }
          label: "line.3.horizontal
                  .decrease.circle" }
      .refreshable { await reload() }
      .sheet(item: $spool) {
          SpoolDetail() } }
  }
  Tab("Settings", systemImage: "gearshape") {
    NavigationStack { SettingsForm() }
      .fullScreenCover { QRScanner() }
  }
}
